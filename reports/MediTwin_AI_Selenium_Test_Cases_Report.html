# Automated Concurrency & Load Test Suite import time import pytest import concurrent.futures import requests BACKEND_URL = "http://localhost:5000/api/auth/login" class TestPatientLoginLoad: def send_login_request(self, user_index): payload = { "email": f"patient.test_{user_index % 50}@meditwin.local", "password": "PatientPass@123" } start_time = time.time() try: # Send concurrent HTTP authentication request response = requests.post(BACKEND_URL, json=payload, timeout=8) duration_ms = (time.time() - start_time) * 1000 return { "user_index": user_index, "status_code": response.status_code, "duration_ms": duration_ms, "success": response.status_code in [200, 401] } except Exception as e: return {"user_index": user_index, "status_code": 0, "success": False, "error": str(e)} def test_load_1000_concurrent_patients(self): total_virtual_users = 1000 concurrency_workers = 100 # 100 concurrent thread pool workers start_overall = time.time() results = [] with concurrent.futures.ThreadPoolExecutor(max_workers=concurrency_workers) as executor: futures = [executor.submit(self.send_login_request, i) for i in range(total_virtual_users)] for future in concurrent.futures.as_completed(futures): results.append(future.result()) total_time = time.time() - start_overall successful_requests = [r for r in results if r["success"]] durations = [r["duration_ms"] for r in successful_requests] avg_latency = sum(durations) / len(durations) if durations else 0 error_rate = ((total_virtual_users - len(successful_requests)) / total_virtual_users) * 100 # Assertions for Load SLA assert len(successful_requests) == 1000, f"Expected 1000 successful logins, got {len(successful_requests)}" assert error_rate == 0.0, f"Error rate exceeded threshold: {error_rate}%" assert avg_latency < 2000, f"Average latency SLA violated (>2000ms): {avg_latency:.1f}ms"
